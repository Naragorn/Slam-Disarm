# OWED, does not pass yet (2026-11-23): not run by tools/harness.sh (only harness/*.txt).
# Run by hand: tools/harness.sh harness/owed/1-slam.txt
# Where it stands: the grab now holds through the slam and the NPC's hand comes down onto the block, but at
# about 1-2 m/s. The game raises no collision event for a ragdoll part that meets the world slower than
# Ragdoll.collisionMinVelocity (2 m/s), and the mod counts from Min Slam Speed (2.5 m/s). The NPC's arm follows
# the player's controller only partly (12 m/s controller -> 2-3 m/s arm at best, 0.12-0.24 m of travel), and the
# player's own hand stays at the NPC's chest while it holds the forearm. See DEVELOPMENT.md, "Harness".
#
# A real slam: the player grabs the NPC's sword forearm and slams it down onto a block twice (BasHarness
# player-grab-npc, place-block, player-slam: real physics, the game's own collision events). The block stands in
# for a table: a static collider without a ColliderGroup, which the mod counts as World. The second slam disarms;
# the hand may not pick the sword up again for No Regrab Time.
# player-eyes first: the mock headset sits 0.16 m above the creature's eyes otherwise, and the game measures the
# grab's reach from the head (PlayerHand drops a grip farther than 1.22 m from it).
# The block goes under the NPC's hand (from=npc-hand), 0.13 m below it: clear of his thigh and of the sword,
# whose blade points from the hand towards the player.
wait-level
set-option SlamDisarm "Slam Disarm Enabled" true
set-option SlamDisarm "Debug Logging" true
set-option SlamDisarm "Slams To Disarm" 2
set-option SlamDisarm "Min Slam Speed" 2.5
set-option SlamDisarm "Slam Window" 3
set-option SlamDisarm "Disarm Chance" 1
set-option SlamDisarm "No Regrab Time" 5
frame-rate 90
player-eyes

# HumanDummy runs no behaviour tree, so it never draws its weapon: put the sword into its hand.
spawn-npc HumanMale distance=0.5 facing=0 brain=HumanDummy container=CivilianMeleeSword
npc-grab SwordShortCommon hand=left
expect-npc holding True within=2
player-grab-npc LeftArm section=Lower
expect-log [SlamDisarm] GRAB |
wait 1
place-block 0,-0.19,0 size=0.12 from=npc-hand
record-start slam fps=30 width=960 height=540 views=side,npc distance=1.8 rise=0.8 angle=90
mark
player-slam 0,-0.6,0 speed=12
expect-log [SlamDisarm] SLAM |
wait 0.5
player-slam 0,-0.6,0 speed=12
await-log 2 [SlamDisarm] DISARM |
wait 1
record-stop
expect-npc holding False within=2
player-release

# No regrab: a sword put into the hand is let go again.
mark
npc-grab SwordShortCommon hand=left
expect-npc holding False within=1
expect-log [SlamDisarm] REGRAB |
expect-no-log Exception
quit
